<script>
(function () {
  const MAX = 100;
  const BEST_KEY = 'guess-number-best';

  const input      = document.getElementById('guessInput');
  const guessBtn   = document.getElementById('guessBtn');
  const restartBtn = document.getElementById('restartBtn');
  const messageEl  = document.getElementById('message');
  const triesEl    = document.getElementById('tries');
  const bestEl     = document.getElementById('best');
  const historyEl  = document.getElementById('historyList');

  let target = 0;
  let tries = 0;
  let over = false;
  let guesses = [];   // [{ value, result: 'high' | 'low' | 'correct' }]

  const randomInt = (max) => Math.floor(Math.random() * max) + 1;

  function loadBest() {
    const v = Number(localStorage.getItem(BEST_KEY));
    return Number.isInteger(v) && v > 0 ? v : null;
  }

  function renderBest() {
    const best = loadBest();
    bestEl.textContent = best ? best + ' 次' : '—';
  }

  function setMessage(text, type) {
    messageEl.textContent = text;
    messageEl.className = 'message ' + (type || 'info');
  }

  function renderHistory() {
    historyEl.innerHTML = '';
    guesses.forEach(function (item) {
      const li = document.createElement('li');
      li.className = item.result;
      const icon = item.result === 'correct' ? '✅'
                 : item.result === 'high'    ? '⬇️ 偏大'
                 :                             '⬆️ 偏小';
      li.innerHTML = '<span>' + item.value + '</span><span>' + icon + '</span>';
      historyEl.appendChild(li);
    });
  }

  function submitGuess() {
    if (over) return;

    const raw = input.value.trim();
    if (raw === '') {
      setMessage('请输入一个数字～', 'warn');
      return;
    }

    const n = Number(raw);
    if (!Number.isInteger(n) || n < 1 || n > MAX) {
      setMessage('请输入 1 到 ' + MAX + ' 之间的整数', 'warn');
      input.select();
      return;
    }

    if (guesses.some(function (g) { return g.value === n; })) {
      setMessage(n + ' 已经猜过啦，换一个吧 😉', 'warn');
      input.select();
      return;
    }

    tries++;
    triesEl.textContent = tries;

    if (n === target) {
      over = true;
      guesses.push({ value: n, result: 'correct' });

      const prevBest = loadBest();
      const isRecord = prevBest === null || tries < prevBest;
      if (isRecord) {
        localStorage.setItem(BEST_KEY, String(tries));
        renderBest();
      }

      setMessage(
        isRecord
          ? '🎉 猜对了！答案就是 ' + target + '，用了 ' + tries + ' 次，刷新最佳成绩！'
          : '🎉 猜对了！答案就是 ' + target + '，你用了 ' + tries + ' 次。',
        'success'
      );

      guessBtn.disabled = true;
      input.disabled = true;
    } else {
      const result = n > target ? 'high' : 'low';
      guesses.push({ value: n, result: result });
      setMessage(n > target ? '太大了 ⬇️ 再小一点' : '太小了 ⬆️ 再大一点', 'info');
      input.value = '';
    }

    renderHistory();
    if (!over) input.focus();
  }

  function newGame() {
    target = randomInt(MAX);
    tries = 0;
    over = false;
    guesses = [];

    triesEl.textContent = '0';
    historyEl.innerHTML = '';
    input.value = '';
    input.disabled = false;
    guessBtn.disabled = false;

    setMessage('新的一局开始啦，数字在 1 ~ ' + MAX + ' 之间，加油！', 'info');
    renderBest();
    input.focus();
  }

  guessBtn.addEventListener('click', submitGuess);
  restartBtn.addEventListener('click', newGame);

  input.addEventListener('keydown', function (e) {
    if (e.key === 'Enter') submitGuess();
  });

  newGame();
})();
</script>
